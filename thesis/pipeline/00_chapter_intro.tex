\section{Understanding the Pipeline}
\label{ch:pipeline}


This chapter walks through the segmentation pipeline for \textbf{CAS-Net} \cite{dong2023casnet}. The best preforming method in Kit study. 

\begin{figure}[p]
  \centering
  \begin{adjustbox}{max width=\linewidth, max height=0.88\textheight, keepaspectratio}
  \begin{tikzpicture}[
      font=\normalsize,
      block/.style={draw, rounded corners, align=center, minimum height=13mm,
                    minimum width=52mm, fill=blue!4, inner sep=2mm},
      io/.style={draw, rounded corners, align=center, minimum height=13mm,
                 minimum width=52mm, fill=gray!10, inner sep=2mm},
      decision/.style={draw, align=center, minimum height=13mm, minimum width=52mm,
                        fill=orange!8, inner sep=2mm},
      arr/.style={-{Latex[length=2.5mm]}, thick},
      tab/.style={font=\bfseries, fill=white, draw, rounded corners,
                  inner sep=2mm, align=left},
    ]
 
    % --- shared prep, spans both lanes ---
    \node[io] (raw) at (0,0) {Raw CCTA scan + GT mask};
    \node[block] (resample) at (0,-2.2) {Resample to 0.5\,mm isotropic};
 
    % --- training lane (left) ---
    \node[block] (crop) at (-4.2,-5.6) {Crop + normalise + augment};
    \node[block] (fwd) at (-4.2,-8.0) {CAS-Net forward pass};
    \node[block] (loss) at (-4.2,-10.4) {Dice loss + backprop};
    \node[decision] (opt) at (-4.2,-12.8) {Adam optimiser \\ \small 1000 epochs};
    \node[io] (ckpt) at (-4.2,-15.2) {Save best checkpoint};
 
    % --- inference lane (right) ---
    \node[block] (norm) at (4.2,-5.6) {Normalise intensity};
    \node[block] (tile) at (4.2,-8.0) {Tiled inference + blend};
    \node[block] (tta) at (4.2,-10.4) {Mirror-flip TTA};
    \node[block] (post) at (4.2,-12.8) {Post-process + resample};
    \node[io] (pred) at (4.2,-15.2) {Save prediction};
 
    % --- evaluation, spans both lanes ---
    \node[block] (eval) at (0,-17.6) {Evaluate vs.\ GT \\ \small Dice, HD95, Betti, clDice};
 
    % --- phase lane backgrounds ---
    \begin{scope}[on background layer]
      \node[draw, dashed, rounded corners, fill=blue!2, inner ysep=6mm, inner xsep=8mm,
            fit=(crop)(fwd)(loss)(opt)(ckpt)] (trainbox) {};
      \node[tab, anchor=south west] at ($(trainbox.north west)+(2mm,-1mm)$)
        {TRAINING -- once, offline};
 
      \node[draw, dashed, rounded corners, fill=orange!2, inner ysep=6mm, inner xsep=8mm,
            fit=(norm)(tile)(tta)(post)(pred)] (inferbox) {};
      \node[tab, anchor=south east] at ($(inferbox.north east)+(-2mm,-1mm)$)
        {INFERENCE -- per scan, test time};

      % drawn behind the crop/norm boxes so the fork reads as entering from beneath them
      \draw[arr] (resample.south) -- ++(0,-0.9) -| (crop.north);
      \draw[arr] (resample.south) -- ++(0,-0.9) -| (norm.north);
    \end{scope}

    % --- shared prep arrows ---
    \draw[arr] (raw) -- (resample);
 
    % --- training chain ---
    \draw[arr] (crop) -- (fwd);
    \draw[arr] (fwd) -- (loss);
    \draw[arr] (loss) -- (opt);
    \draw[arr] (opt) -- (ckpt);
    \draw[arr] (opt.west) -- ++(-1.3,0) |- node[pos=0.25, left, font=\scriptsize, align=center]{repeat\\each epoch} (crop.west);
 
    % --- checkpoint feeds inference (the one link between lanes) ---
    \draw[arr, ultra thick] (ckpt.east) -- ++(1.3,0) |- node[pos=0.5, above, font=\small, fill=white, inner sep=1pt]{frozen weights} (tile.west);
 
    % --- inference chain ---
    \draw[arr] (norm) -- (tile);
    \draw[arr] (tile) -- (tta);
    \draw[arr] (tta) -- (post);
    \draw[arr] (post) -- (pred);
 
    % --- evaluation: only inference's prediction is scored, against GT ---
    \draw[arr] (pred.south) -- ++(0,-0.9) -| (eval.east);
 
  \end{tikzpicture}
  \end{adjustbox}
  \caption{The segmentation pipeline. Training (left lane) runs once, offline; inference (right lane) runs per scan at test time using the frozen checkpoint produced by training. Inference's resulting predictions are compared against ground truth in the final evaluation step.}
  \label{fig:pipeline-overview}
\end{figure}
