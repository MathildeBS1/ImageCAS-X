\section{Understanding the Pipeline}
\label{ch:pipeline}


This chapter walks through the segmentation pipeline for \textbf{CAS-Net} \cite{dong2023casnet}. The best preforming method in Kit study. 

\begin{figure}[htbp]
  \centering
  % shrink only if wider than the text block; graphicx alone, no adjustbox needed
  \resizebox{\ifdim\width>\linewidth\linewidth\else\width\fi}{!}{%
  \begin{tikzpicture}[
      font=\small,
      box/.style={draw, rounded corners, align=center, text width=42mm,
                  minimum height=9mm, inner sep=1.5mm},
      block/.style={box, fill=blue!4},
      wide/.style={text width=52mm},
      io/.style={box, fill=gray!10},
      decision/.style={box, fill=orange!8, sharp corners},
      arr/.style={-{Latex[length=2mm]}, thick},
      tab/.style={font=\footnotesize, fill=white, draw, rounded corners,
                  inner sep=1.2mm, align=left},
    ]

    % --- shared prep, spans both lanes ---
    \node[io, wide] (raw) at (0,0) {Raw CCTA scan + GT mask};
    \node[block, wide] (resample) at (0,-1.3) {Resample to 0.5\,mm isotropic};

    % --- training lane (left) ---
    \node[block] (crop) at (-3.6,-4.0) {Crop + normalise \\ + augment};
    \node[block] (fwd) at (-3.6,-5.3) {CAS-Net forward pass};
    \node[block] (loss) at (-3.6,-6.6) {Dice loss + backprop};
    \node[decision] (opt) at (-3.6,-7.9) {Adam optimiser \\ \footnotesize 1000 epochs};
    \node[io] (ckpt) at (-3.6,-9.2) {Save best checkpoint};

    % --- inference lane (right) ---
    \node[block] (norm) at (3.6,-4.0) {Normalise intensity};
    \node[block] (tile) at (3.6,-5.3) {Tiled inference + blend};
    \node[block] (tta) at (3.6,-6.6) {Mirror-flip TTA};
    \node[block] (post) at (3.6,-7.9) {Post-process + resample};
    \node[io] (pred) at (3.6,-9.2) {Save prediction};

    % --- evaluation, spans both lanes ---
    \node[block, wide] (eval) at (0,-11.0) {Evaluate vs.\ GT \\ \footnotesize Dice, HD95, Betti, clDice};

    % --- phase lane backgrounds (extra outer margin holds the repeat loop and the tabs) ---
    \coordinate (trainpad) at ($(crop.west)+(-7mm,0)$);
    \coordinate (inferpad) at ($(norm.east)+(7mm,0)$);
    \begin{scope}[on background layer]
      \node[draw, dashed, rounded corners, fill=blue!2, inner ysep=4mm, inner xsep=4mm,
            fit=(crop)(fwd)(loss)(opt)(ckpt)(trainpad)] (trainbox) {};
      \node[draw, dashed, rounded corners, fill=orange!2, inner ysep=4mm, inner xsep=4mm,
            fit=(norm)(tile)(tta)(post)(pred)(inferpad)] (inferbox) {};
    \end{scope}

    % lane tabs sit on top of the outer corners, clear of the fork arrows
    \node[tab, anchor=south west] at ($(trainbox.north west)+(2mm,-0.4pt)$)
      {\textbf{TRAINING}\\once, offline};
    \node[tab, anchor=south east] at ($(inferbox.north east)+(-2mm,-0.4pt)$)
      {\textbf{INFERENCE}\\per scan, test time};

    % --- shared prep arrows ---
    \draw[arr] (raw) -- (resample);
    \draw[arr] (resample.south) -- ++(0,-0.4) -| (crop.north);
    \draw[arr] (resample.south) -- ++(0,-0.4) -| (norm.north);

    % --- training chain ---
    \draw[arr] (crop) -- (fwd);
    \draw[arr] (fwd) -- (loss);
    \draw[arr] (loss) -- (opt);
    \draw[arr] (opt) -- (ckpt);
    \draw[arr] (opt.west) -- ++(-0.55,0) |- node[pos=0.25, sloped, above, font=\scriptsize]{repeat each epoch} (crop.west);

    % --- checkpoint feeds inference (the one link between lanes) ---
    \draw[arr, very thick] (ckpt.east) -- (0,-9.2) |- node[pos=0.25, sloped, above, font=\scriptsize]{frozen weights} (tile.west);

    % --- inference chain ---
    \draw[arr] (norm) -- (tile);
    \draw[arr] (tile) -- (tta);
    \draw[arr] (tta) -- (post);
    \draw[arr] (post) -- (pred);

    % --- evaluation: only inference's prediction is scored, against GT ---
    \draw[arr] (pred.south) -- ++(0,-0.9) -| (eval.east);

  \end{tikzpicture}%
  }
  \caption{The segmentation pipeline. Training (left lane) runs once, offline; inference (right lane) runs per scan at test time using the frozen checkpoint produced by training. Inference's resulting predictions are compared against ground truth in the final evaluation step.}
